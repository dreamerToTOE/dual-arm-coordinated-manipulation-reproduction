cmake_minimum_required(VERSION 3.16)
project(task01_rear_handoff)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(task01_rear_handoff task01_rear_handoff.cpp)
target_compile_features(task01_rear_handoff PRIVATE cxx_std_17)
target_compile_options(task01_rear_handoff PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(task01_rear_handoff rclcpp moveit_core
  moveit_ros_planning_interface moveit_msgs sensor_msgs std_msgs geometry_msgs
  shape_msgs trajectory_msgs)
target_link_libraries(task01_rear_handoff yaml-cpp nlohmann_json::nlohmann_json)
